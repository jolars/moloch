\section{Introduction}

\begin{frame*}
  \frametitle{Moloch}

  The Moloch theme has minimal visual noise. It is a fork of the
  \href{https://github.com/matze/mtheme}{Metropolis theme} by Matthias Vogelgesang,
  which in turn was inspired by the
  \href{https://github.com/hsrmbeamertheme/hsrmbeamertheme}{HSRM theme}
  by Benjamin Weiss. \medskip

  Enable the theme for \texttt{ltx-talk} by calling
  \begin{verbatim}
\DocumentMetadata{lang=en-US,tagging=on}
\documentclass{ltx-talk}
\usepackage{moloch-talk}
  \end{verbatim}
\end{frame*}

\begin{frame*}
  \frametitle{Sections}
  Sections group slides of the same topic by introducing a section page between
  them. A progress bar indicates how far along in the presentation you are.

  \begin{verbatim}
\section{Title Customization}
  \end{verbatim}

  Place section commands between frames. Section pages count toward the frame
  number in \texttt{ltx-talk}.
\end{frame*}

\section{Title Customization}

\begin{frame*}
  \frametitle{Moloch Frame Titles}
  Use the class's template interface to change the font of frame titles:
  \begin{verbatim}
\EditInstance{frametitle}{header}{
  font=\Large\bfseries
}
  \end{verbatim}

  The following slides show two font choices. The title band has a fixed height,
  so wrapped titles may need a taller band.
\end{frame*}

  \end{block}

\begin{frame}
  \frametitle{Progress Bars}
  Progress bars visualize the progress of the presentation. \medskip

  By default, bars appear on section pages. Choose
  \texttt{sectionpage=progressbar}, \texttt{simple}, or \texttt{none} to change
  this. \medskip

  Enable progress bars on frames with
  \texttt{progressbar=none}, \texttt{head}, \texttt{frametitle}, or \texttt{foot}.
\end{frame}

  \begin{verbatim}

  titleseparator,author,institute,date}]
